-- Not a real migration file yet (renamed .sql.txt so drizzle-kit doesn't pick
-- it up before you've run `pnpm db:generate` once). After generating the
-- first real migration from schema.ts, append this trigger to it, or add it
-- as its own follow-up migration -- it enforces BUILD.md §0 rule 3 and §4.1:
-- "signal is append-only." Drizzle's schema builder cannot express triggers,
-- so this has to be hand-written SQL either way.

CREATE OR REPLACE FUNCTION signal_is_append_only() RETURNS trigger AS $$
BEGIN
  RAISE EXCEPTION 'signal is append-only: % attempted on signal.id=%', TG_OP, OLD.id;
END; $$ LANGUAGE plpgsql;

CREATE TRIGGER signal_no_mutate
  BEFORE UPDATE OR DELETE ON signal
  FOR EACH ROW EXECUTE FUNCTION signal_is_append_only();

-- Also enforce the CHAR(2) constraints Drizzle's `text()` can't express:
ALTER TABLE place ALTER COLUMN country_code TYPE char(2);
ALTER TABLE place_name ALTER COLUMN lang TYPE char(2);
